Container Selection Guide

Container Mental Model Best For Time Complexity
vector<T> Dynamic array Default choice, random access O(1) access, O(n) insert
string Char array Text, sliding window O(1) access
unordered_map<K,V> Hash table Two Sum, frequency counts O(1) avg lookup
unordered_set<T> Hash set Membership test, visited O(1) avg lookup
map<K,V> / set<T> Sorted tree Ordered data, intervals O(log n) operations
stack<T> Plates (LIFO) Parentheses, monotonic stack O(1) push/pop
queue<T> Line (FIFO) BFS, level traversal O(1) push/pop
deque<T> Double-ended Sliding window max O(1) both ends
priority_queue<T> Heap Top-K, Dijkstra O(log n) insert

Critical Syntax Rules

std::X              // X from standard library
T&                  // Reference (modifies original)
const T&            // Read-only reference (no copy)
auto                // Type inferred from RHS
begin(), end()      // Iterator range [begin, end)
[](...) { ... }     // Lambda (inline function)

References vs Pointers

void demo() {
    int value = 42;
    
    // Reference (alias)
    int& ref = value;
    ref = 100;  // Changes value to 100
    
    // Pointer (stores address)
    int* ptr = &value;
    *ptr = 200;  // Dereference and change
}

Static Keyword (Two Meanings)

Important: Static has different meanings depending on context. 
// 1. File scope - limits visibility
static void helper() {
    // Only visible in this .cpp file
}

// 2. Local variable - persists across calls
int generateId() {
    static int id = 0;  // Initialized once
    return ++id;        // Returns 1, 2, 3...
}

Inheritance and Polymorphism

// Abstract base class
class Shape {
protected:
    std::string color;
    
public:
    Shape(std::string c) : color(c) {}
    virtual ~Shape() {}
    
    // Pure virtual (abstract)
    virtual double area() const = 0;
    
    virtual void draw() const {
        std::cout << "Drawing " << color << " shape\n";
    }
};

class Circle : public Shape {
private:
    double radius;
    
public:
    Circle(std::string c, double r) : Shape(c), radius(r) {}
    
    double area() const override {
        return 3.14159 * radius * radius;
    }
};

Pattern 1: Two Sum (Hash Map)

LeetCode 1
class Solution {
public:
    std::vector<int> twoSum(std::vector<int>& nums, int target) {
        std::unordered_map<int, int> seen;
        
        for(int i = 0; i < nums.size(); i++) {
            int complement = target - nums[i];
            
            if(seen.count(complement)) {
                return {seen[complement], i};
            }
            seen[nums[i]] = i;
        }
        return {};
    }
};
Pattern: Check complement → store current
Key: Unordered map for O(1) lookups

Pattern 2: Valid Parentheses (Stack)

LeetCode 20
class Solution {
public:
    bool isValid(std::string s) {
        std::stack<char> stk;
        
        for(char c : s) {
            if(c == '(' || c == '{' || c == '[') {
                stk.push(c);
            } else {
                if(stk.empty()) return false;
                char top = stk.top();
                stk.pop();
                
                if((c == ')' && top != '(') ||
                   (c == '}' && top != '{') ||
                   (c == ']' && top != '['))
                    return false;
            }
        }
        return stk.empty();
    }
};
Pattern: LIFO matching

Pattern 3: BFS on Grid

LeetCode 200 - Number of Islands
int numIslands(vectorchar>>& grid) {
    int rows = grid.size();
    int cols = grid[0].size();
    int islands = 0;
    
    int dr[] = {1, -1, 0, 0};
    int dc[] = {0, 0, 1, -1};
    
    for(int r = 0; r < rows; r++) {
        for(int c = 0; c < cols; c++) {
            if(grid[r][c] != '1') continue;
            
            islands++;
            grid[r][c] = '0';  // Mark visited
            
            queueint,int>> q;
            q.push({r, c});
            
            while(!q.empty()) {
                auto [cr, cc] = q.front();
                q.pop();
                
                for(int k = 0; k < 4; k++) {
                    int nr = cr + dr[k];
                    int nc = cc + dc[k];
                    
                    if(nr < 0 || nr >= rows || nc < 0 || nc >= cols)
                        continue;
                    if(grid[nr][nc] != '1') continue;
                    
                    grid[nr][nc] = '0';
                    q.push({nr, nc});
                }
            }
        }
    }
    return islands;
}
Pattern: BFS + mark visited
Key: Queue for level-by-level exploration

Pattern 4: Top K Frequent (Heap)

LeetCode 347
vector<int> topKFrequent(vector<int>& nums, int k) {
    // Count frequencies
    unordered_map<int, int> freq;
    for(int x : nums) freq[x]++;
    
    // Min-heap of (frequency, value)
    using P = pair<int, int>;
    auto cmp = [](const P& a, const P& b) {
        return a.first > b.first;
    };
    priority_queue, decltype(cmp)> heap(cmp);
    
    for(const auto& [val, f] : freq) {
        heap.push({f, val});
        if(heap.size() > k) heap.pop();
    }
    
    vector<int> result;
    while(!heap.empty()) {
        result.push_back(heap.top().second);
        heap.pop();
    }
    return result;
}
Pattern: Keep best K using min-heap

Pattern 5: Sliding Window Maximum

LeetCode 239
vector<int> maxSlidingWindow(vector<int>& nums, int k) {
    deque<int> dq;  // Stores indices
    vector<int> result;
    
    for(int i = 0; i < nums.size(); i++) {
        // Remove indices outside window
        if(!dq.empty() && dq.front() <= i - k)
            dq.pop_front();
        
        // Maintain decreasing order
        while(!dq.empty() && nums[dq.back()] <= nums[i])
            dq.pop_back();
        
        dq.push_back(i);
        
        if(i >= k - 1)
            result.push_back(nums[dq.front()]);
    }
    return result;
}
Pattern: Monotonic deque
🧠 Memory Management ▼

Smart Pointers

// unique_ptr - Single ownership
std::unique_ptr p1 = std::make_unique();
p1->use();
std::unique_ptr p2 = std::move(p1);  // Transfer

// shared_ptr - Shared ownership
std::shared_ptr s1 = std::make_shared();
{
    std::shared_ptr s2 = s1;
    std::cout << s1.use_count() << "\n";  // 2
}
std::cout << s1.use_count() << "\n";  // 1

// weak_ptr - Non-owning reference
std::weak_ptr w = s1;
if(auto locked = w.lock()) {
    locked->use();
}
💡 Tip: Always prefer smart pointers over raw pointers. They provide automatic memory management and prevent memory leaks.

Singleton Pattern

class Database {
public:
    static Database& getInstance() {
        static Database instance;
        return instance;
    }
    
    Database(const Database&) = delete;
    Database& operator=(const Database&) = delete;
    
private:
    Database() {}
};

Factory Pattern

class VehicleFactory {
public:
    enum Type { CAR, MOTORCYCLE };
    
    static std::unique_ptr create(Type type) {
        switch(type) {
            case CAR: return std::make_unique();
            case MOTORCYCLE: return std::make_unique();
        }
    }
};

Common STL Algorithms

std::vector<int> v = {3, 1, 4, 1, 5};

// Sorting
std::sort(v.begin(), v.end());

// Binary search (requires sorted)
auto it = std::lower_bound(v.begin(), v.end(), 4);
int idx = it - v.begin();  // Iterator to index

// Finding
auto found = std::find(v.begin(), v.end(), 5);
if(found != v.end()) {
    std::cout << "Found!\n";
}

// Min/Max
auto minIt = std::min_element(v.begin(), v.end());
auto maxIt = std::max_element(v.begin(), v.end());

// Count
int count = std::count(v.begin(), v.end(), 1);

// Accumulate (sum)
int sum = std::accumulate(v.begin(), v.end(), 0);

// Transform
std::transform(v.begin(), v.end(), v.begin(),
               [](int x) { return x * 2; });

// Remove + erase idiom
v.erase(std::remove(v.begin(), v.end(), 2), v.end());
